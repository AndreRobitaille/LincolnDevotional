# Entry Pages for SEO and Crawlability Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Generate one static, crawlable HTML page per devotional entry plus sitemap support, while keeping the current homepage as the JS-driven “today” experience.

**Architecture:** Add a Python generator that reads `data/entries.json` and `data/esv_cache.json`, writes `entries/<month-day>/index.html`, and emits `sitemap.xml` plus `robots.txt`. Keep `index.html` as the interactive homepage, but add a stable permalink to each generated entry page and reuse existing CSS/theme assets on generated pages.

**Tech Stack:** Static HTML, vanilla JavaScript, CSS, Python 3 standard library (`json`, `html`, `pathlib`, `xml.etree.ElementTree`, `unittest`)

from __future__ import annotations

from html import escape
import json
from pathlib import Path
from xml.etree import ElementTree as ET


MONTH_NAMES = {
    1: "january",
    2: "february",
    3: "march",
    4: "april",
    5: "may",
    6: "june",
    7: "july",
    8: "august",
    9: "september",
    10: "october",
    11: "november",
    12: "december",
}

ROOT = Path(__file__).resolve().parent.parent
ENTRIES_PATH = ROOT / "data" / "entries.json"
ESV_CACHE_PATH = ROOT / "data" / "esv_cache.json"
OUTPUT_ROOT = ROOT
SITE_URL = "https://lincolndevotional.com"
REQUIRED_FIELDS = ("mmdd", "month", "day", "display_date", "title", "bible_verse", "verse_ref", "poem")


def load_json(path: Path):
    with path.open(encoding="utf-8") as handle:
        return json.load(handle)


def slugify_entry(entry):
    month_name = MONTH_NAMES[entry["month"]]
    return f"{month_name}-{entry['day']}"


def build_entry_href(entry):
    return f"/entries/{slugify_entry(entry)}/"


def build_description(entry):
    title = entry["title"].strip()
    date_text = entry["display_date"].strip()
    verse_ref = entry["verse_ref"].strip()
    return f"{date_text}: {title}. Scripture from {verse_ref} with devotional poem from The Believer's Daily Treasure."


def normalize_poem_lines(poem_text):
    return [line.replace("\r", "") for line in poem_text.splitlines()]


def validate_entries(entries):
    seen_slugs = set()
    for entry in entries:
        missing = [field for field in REQUIRED_FIELDS if not entry.get(field)]
        if missing:
            raise ValueError(f"Entry {entry.get('mmdd', '<unknown>')} missing required fields: {', '.join(missing)}")

        slug = slugify_entry(entry)
        if slug in seen_slugs:
            raise ValueError(f"Duplicate slug generated: {slug}")
        seen_slugs.add(slug)
```

def render_poem_html(poem_text):
    poem_lines = []
    for line in normalize_poem_lines(poem_text):
        class_name = "poem-line poem-line--blank" if not line.strip() else "poem-line"
        poem_lines.append(f'<div class="{class_name}">{escape(line)}</div>')
    return "\n".join(poem_lines)


def render_esv_block(esv_text):
    if not esv_text:
        return ""
    return f"""
              <div class="verse-block">
                <span class="version-label">ESV</span>
                <p class="entry-text">{escape(esv_text)}</p>
              </div>"""
```

def render_entry_page(entry, esv_text, previous_entry, next_entry, site_url):
    href = build_entry_href(entry)
    canonical_url = f"{site_url}{href}"
    description = escape(build_description(entry))
    previous_href = build_entry_href(previous_entry) if previous_entry else "/"
    next_href = build_entry_href(next_entry) if next_entry else "/"
    esv_block = render_esv_block(esv_text)

    return f"""<!doctype html>
<html lang="en">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>{escape(entry['title'])} | The Believer's Daily Treasure</title>
    <link rel="canonical" href="{canonical_url}" />
    <meta name="description" content="{description}" />
    <meta property="og:title" content="{escape(entry['title'])} | The Believer's Daily Treasure" />
    <meta property="og:description" content="{description}" />
    <meta property="og:url" content="{canonical_url}" />
    <link rel="preconnect" href="https://fonts.googleapis.com" />
    <link rel="preconnect" href="https://fonts.gstatic.com" crossorigin />
    <link href="https://fonts.googleapis.com/css2?family=Crimson+Pro:wght@400;500;600&family=Newsreader:wght@400;500;600&display=swap" rel="stylesheet" />
    <link rel="stylesheet" href="/style.css?v=20260509" />
  </head>
  <body>
    <div class="page">
      <header class="site-header">
        <div class="brand">
          <p class="site-eyebrow">Abraham Lincoln's Daily Devotional</p>
          <h1 class="site-title">The Believer's Daily Treasure</h1>
          <p class="site-tagline">Texts of scripture, arranged for every day in the year.</p>
        </div>
        <div class="site-actions">
          <nav class="site-nav" aria-label="Primary">
            <a href="/">Devotional</a>
            <a href="/about.html">About</a>
          </nav>
          <button class="theme-toggle" id="themeToggle" type="button">Dark mode</button>
        </div>
      </header>

      <main class="main-content">
        <nav class="entry-page-nav" aria-label="Entry navigation">
          <a class="nav-button nav-button--link" href="{previous_href}">&larr; Previous</a>
          <a class="entry-permalink" href="/">Today&apos;s devotional</a>
          <a class="nav-button nav-button--link" href="{next_href}">Next &rarr;</a>
        </nav>

        <article class="entry-card">
          <header class="entry-header">
            <p class="entry-date">{escape(entry['display_date'])}</p>
            <h2 class="entry-title">{escape(entry['title'])}</h2>
          </header>
          <section class="entry-section entry-section--scripture">
            <h3 class="entry-section-title">Scripture</h3>
            <div class="verse-columns">
              <div class="verse-block">
                <span class="version-label">KJV</span>
                <p class="entry-text">{escape(entry['bible_verse'])}</p>
              </div>{esv_block}
            </div>
            <p class="entry-verse-ref">{escape(entry['verse_ref'])}</p>
          </section>
          <section class="entry-section entry-poem">
            <h3 class="entry-section-title">Poem</h3>
            <div class="entry-text">{render_poem_html(entry['poem'])}</div>
          </section>
        </article>
      </main>

      <footer class="site-footer">
        <p class="footer-sites"><a href="https://lincolndevotional.com/">LincolnDevotional.com</a>, the daily devotional Abraham Lincoln carried.</p>
        <p class="footer-sites"><a href="https://tworiversmatters.com/">TwoRiversMatters.com</a>, covering Two Rivers, Wisconsin city government, meetings, and civic news.</p>
        <p class="footer-legal"><a href="/copyright.html">Copyright</a></p>
      </footer>
    </div>

    <script src="/theme.js?v=20260123"></script>
  </body>
</html>
"""


def write_sitemap(entries, output_root, site_url):
    urlset = ET.Element("urlset", xmlns="http://www.sitemaps.org/schemas/sitemap/0.9")
    for url in ["/", "/about.html", "/copyright.html", *[build_entry_href(entry) for entry in entries]]:
        url_element = ET.SubElement(urlset, "url")
        location = ET.SubElement(url_element, "loc")
        location.text = f"{site_url}{url}"
    ET.ElementTree(urlset).write(output_root / "sitemap.xml", encoding="utf-8", xml_declaration=True)


def write_robots_txt(output_root, site_url):
    robots_path = output_root / "robots.txt"
    robots_path.write_text(f"User-agent: *\nAllow: /\n\nSitemap: {site_url}/sitemap.xml\n", encoding="utf-8")


def generate_site(entries, esv_cache, output_root, site_url):
    validate_entries(entries)
    output_root.mkdir(parents=True, exist_ok=True)

    for index, entry in enumerate(entries):
        previous_entry = entries[index - 1] if index > 0 else entries[-1]
        next_entry = entries[index + 1] if index < len(entries) - 1 else entries[0]
        esv_text = (esv_cache.get(entry["mmdd"]) or {}).get("text")
        html = render_entry_page(entry, esv_text, previous_entry, next_entry, site_url)

        page_dir = output_root / "entries" / slugify_entry(entry)
        page_dir.mkdir(parents=True, exist_ok=True)
        (page_dir / "index.html").write_text(html, encoding="utf-8")

    write_sitemap(entries, output_root, site_url)
    write_robots_txt(output_root, site_url)


def main():
    entries = load_json(ENTRIES_PATH)
    esv_cache = load_json(ESV_CACHE_PATH)
    generate_site(entries, esv_cache, OUTPUT_ROOT, SITE_URL)
    print(f"Generated {len(entries)} entry pages, sitemap.xml, and robots.txt")


if __name__ == "__main__":
    main()

        return `${entry.display_date.toLowerCase().replace(/\s+/g, '-')}`;

    function buildEntryHref(entry) {
        return `entries/${slugifyEntry(entry)}/`;

        if (!devotionalContent.permalink) {
